# texture=TEXTURE_NAME # name of texture (assets/sprites/blocks/dirt.png -> dirt) or empty for none. defaults to furniture name
# texture_size=NUMBER # texture size of a single block. default is 8
# size=NUMBER,NUMBER # size of furniture (supported types: sapling,table,chair,door)
# face_player=true/false # should furniture face player when placing (supported types: sapling,table,chair,door)
# drop_table=DROP_TABLE_NAME # name of drop table (assets/config/drop_tables.txt). defaults to no drop
# break_speed=NUMBER # time to break
# tool_power=NUMBER # tool power required for drop
# tool_power_required=true/false # whether or not tool power is required for breaking the furniture at all
# tool_type=pickaxe/axe/hammer # type of tool required to break furniture. furniure breakable with pickaxe are also breakable by hand
